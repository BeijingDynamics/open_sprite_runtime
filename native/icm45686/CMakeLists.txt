cmake_minimum_required(VERSION 3.16)
project(sprite_icm45686_candidate LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(SPRITE_IMU_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../src" CACHE PATH "Existing validated six-axis estimator headers")
set(TDK ../third_party/tdk_icm45686)
add_library(tdk_icm STATIC ${TDK}/imu/inv_imu_driver.c ${TDK}/imu/inv_imu_transport.c)
target_include_directories(tdk_icm PUBLIC ${TDK})
add_executable(sprite_icm45686 main.cpp)
target_include_directories(sprite_icm45686 PRIVATE ${SPRITE_IMU_INCLUDE_DIR})
target_link_libraries(sprite_icm45686 PRIVATE tdk_icm Threads::Threads)
target_compile_options(sprite_icm45686 PRIVATE -Wall -Wextra -Werror)
add_executable(icm_offline_test test.cpp)
target_include_directories(icm_offline_test PRIVATE ${SPRITE_IMU_INCLUDE_DIR})
target_link_libraries(icm_offline_test PRIVATE tdk_icm)
enable_testing()
add_test(NAME icm_offline COMMAND icm_offline_test)
add_executable(icm_packet_test packet_test.cpp)
target_include_directories(icm_packet_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../src)
target_link_libraries(icm_packet_test PRIVATE Threads::Threads)
add_test(NAME icm_packet COMMAND icm_packet_test)
