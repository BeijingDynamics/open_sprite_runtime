cmake_minimum_required(VERSION 3.16)
project(open_sprite_runtime_native LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(sprite_rt_timing src/sprite_rt_timing.cpp)
target_compile_options(sprite_rt_timing PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(sprite_rt_timing PRIVATE pthread)

add_executable(sprite_can_shadow src/sprite_can_shadow.cpp)
target_compile_options(sprite_can_shadow PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(sprite_can_shadow PRIVATE pthread)

enable_testing()
add_executable(test_policy_core tests/test_policy_core.cpp)
target_compile_options(test_policy_core PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
add_test(NAME policy_core COMMAND test_policy_core)

# Offline actor replay only; the installed CAN executable is not replaced.
set(SPRITE_ORT_INCLUDE "" CACHE PATH "ONNX Runtime C++ headers")
set(SPRITE_ORT_LIBRARY "" CACHE FILEPATH "ONNX Runtime shared library")
if(SPRITE_ORT_INCLUDE AND SPRITE_ORT_LIBRARY)
  add_executable(sprite_policy_replay src/sprite_policy_replay.cpp)
  target_include_directories(sprite_policy_replay PRIVATE "${SPRITE_ORT_INCLUDE}")
  target_link_libraries(sprite_policy_replay PRIVATE "${SPRITE_ORT_LIBRARY}")
  target_compile_options(sprite_policy_replay PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
add_executable(sprite_imu_bno src/sprite_imu_bno.cpp)
target_compile_options(sprite_imu_bno PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(test_imu_packet tests/test_imu_packet.cpp)
target_compile_options(test_imu_packet PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME imu_packet COMMAND test_imu_packet)
add_executable(test_bounded_preview tests/test_bounded_preview.cpp)
target_compile_options(test_bounded_preview PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(test_bounded_preview PRIVATE pthread)
add_test(NAME bounded_preview COMMAND test_bounded_preview)
